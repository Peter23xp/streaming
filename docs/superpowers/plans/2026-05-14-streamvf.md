**Goal:** Build a full-stack French streaming site with TMDB data, VidSrc embed player, optional auth, and Neon PostgreSQL.

**Architecture:** Next.js 15 App Router with React Server Components for all content pages; Client Components scoped to interactive islands (player, filters, search, watchlist). TMDB called server-side only; VidSrc iframe lazy-loaded on first click.

**Tech Stack:** Next.js 15, TypeScript, TailwindCSS, shadcn/ui, TMDB API v3, VidSrc, Zustand, Prisma, Neon PostgreSQL, NextAuth.js, Vercel

```ts
import { clsx, type ClassValue } from 'clsx'
import { twMerge } from 'tailwind-merge'

export function cn(...inputs: ClassValue[]) {
  return twMerge(clsx(inputs))
}

export function slugify(text: string): string {
  return text
    .toLowerCase()
    .normalize('NFD')
    .replace(/[̀-ͯ]/g, '')
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/(^-|-$)/g, '')
}

export function formatYear(dateStr: string | undefined): string {
  if (!dateStr) return ''
  return dateStr.slice(0, 4)
}

export function formatRuntime(minutes: number | undefined): string {
  if (!minutes) return ''
  const h = Math.floor(minutes / 60)
  const m = minutes % 60
  return h > 0 ? `${h}h ${m}min` : `${m}min`
}

export function getImageUrl(path: string | null, size: string): string {
  if (!path) return '/placeholder-poster.jpg'
  return `${process.env.NEXT_PUBLIC_TMDB_IMAGE_BASE || 'https://image.tmdb.org/t/p'}/${size}${path}`
}

export function isVF(originalLanguage: string): boolean {
  return originalLanguage === 'fr'
}
```

```ts
// lib/tmdb.ts
import type {
  TMDBMovie, TMDBTVShow, TMDBPaginatedResponse,
  TMDBSeason, DiscoverFilters
} from '@/types/tmdb'

const BASE = process.env.TMDB_BASE_URL!
const KEY = process.env.TMDB_API_KEY!
const LANG = 'fr-FR'
const CACHE = { next: { revalidate: 86400 } }

function url(path: string, params: Record<string, string | number> = {}): string {
  const p = new URLSearchParams({ api_key: KEY, language: LANG, ...Object.fromEntries(Object.entries(params).map(([k,v]) => [k, String(v)])) })
  return `${BASE}${path}?${p}`
}

export async function getTrending(type: 'movie' | 'tv', timeWindow: 'day' | 'week' = 'week'): Promise<TMDBPaginatedResponse<TMDBMovie & TMDBTVShow>> {
  const res = await fetch(url(`/trending/${type}/${timeWindow}`), CACHE)
  return res.json()
}

export async function getPopular(type: 'movie' | 'tv', page = 1): Promise<TMDBPaginatedResponse<TMDBMovie & TMDBTVShow>> {
  const res = await fetch(url(`/${type}/popular`, { page }), CACHE)
  return res.json()
}

export async function getNowPlaying(page = 1): Promise<TMDBPaginatedResponse<TMDBMovie>> {
  const res = await fetch(url('/movie/now_playing', { page }), CACHE)
  return res.json()
}

export async function getTopRated(type: 'movie' | 'tv', page = 1): Promise<TMDBPaginatedResponse<TMDBMovie & TMDBTVShow>> {
  const res = await fetch(url(`/${type}/top_rated`, { page }), CACHE)
  return res.json()
}

export async function getMovieDetail(id: number): Promise<TMDBMovie> {
  const res = await fetch(url(`/movie/${id}`, { append_to_response: 'credits,videos,similar' }), CACHE)
  return res.json()
}

export async function getTVDetail(id: number): Promise<TMDBTVShow> {
  const res = await fetch(url(`/tv/${id}`, { append_to_response: 'credits,videos,similar' }), CACHE)
  return res.json()
}

export async function getTVSeason(id: number, seasonNumber: number): Promise<TMDBSeason> {
  const res = await fetch(url(`/tv/${id}/season/${seasonNumber}`), CACHE)
  return res.json()
}

export async function searchMulti(query: string, page = 1) {
  const res = await fetch(url('/search/multi', { query, page }), { next: { revalidate: 60 } })
  return res.json()
}

export async function discoverByGenre(
  type: 'movie' | 'tv',
  genreId: number,
  filters: DiscoverFilters = {}
): Promise<TMDBPaginatedResponse<TMDBMovie & TMDBTVShow>> {
  const params: Record<string, string | number> = {
    with_genres: genreId,
    sort_by: filters.sort_by ?? 'popularity.desc',
    page: filters.page ?? 1,
  }
  if (filters.year) params['primary_release_year'] = filters.year
  if (filters.with_origin_country) params['with_origin_country'] = filters.with_origin_country
  const res = await fetch(url(`/discover/${type}`, params), CACHE)
  return res.json()
}

export async function discoverMovies(filters: DiscoverFilters & { with_genres?: number } = {}): Promise<TMDBPaginatedResponse<TMDBMovie>> {
  const params: Record<string, string | number> = {
    sort_by: filters.sort_by ?? 'popularity.desc',
    page: filters.page ?? 1,
  }
  if (filters.with_genres) params['with_genres'] = filters.with_genres
  if (filters.year) params['primary_release_year'] = filters.year
  if (filters.with_origin_country) params['with_origin_country'] = filters.with_origin_country
  const res = await fetch(url('/discover/movie', params), CACHE)
  return res.json()
}

export async function discoverTV(filters: DiscoverFilters & { with_genres?: number; status?: string } = {}): Promise<TMDBPaginatedResponse<TMDBTVShow>> {
  const params: Record<string, string | number> = {
    sort_by: filters.sort_by ?? 'popularity.desc',
    page: filters.page ?? 1,
  }
  if (filters.with_genres) params['with_genres'] = filters.with_genres
  if (filters.year) params['first_air_date_year'] = filters.year
  if (filters.with_origin_country) params['with_origin_country'] = filters.with_origin_country
  if (filters.status) params['with_status'] = filters.status
  const res = await fetch(url('/discover/tv', params), CACHE)
  return res.json()
}

export async function getGenres(type: 'movie' | 'tv'): Promise<{ genres: { id: number; name: string }[] }> {
  const res = await fetch(url(`/genre/${type}/list`), CACHE)
  return res.json()
}
```

```ts
// store/watchlistStore.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

export interface WatchlistItem {
  id: number
  type: 'movie' | 'tv'
  title: string
  poster: string | null
  year: string
}

interface WatchlistStore {
  items: WatchlistItem[]
  add: (item: WatchlistItem) => void
  remove: (id: number, type: 'movie' | 'tv') => void
  has: (id: number, type: 'movie' | 'tv') => boolean
  clear: () => void
}

export const useWatchlistStore = create<WatchlistStore>()(
  persist(
    (set, get) => ({
      items: [],
      add: (item) =>
        set((s) => ({
          items: s.items.some((i) => i.id === item.id && i.type === item.type)
            ? s.items
            : [...s.items, item],
        })),
      remove: (id, type) =>
        set((s) => ({ items: s.items.filter((i) => !(i.id === id && i.type === type)) })),
      has: (id, type) => get().items.some((i) => i.id === id && i.type === type),
      clear: () => set({ items: [] }),
    }),
    { name: 'streamvf-watchlist' }
  )
)
```

```ts
// app/api/comments/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { auth } from '@/lib/auth'

export async function GET(req: NextRequest) {
  const { searchParams } = new URL(req.url)
  const mediaId = parseInt(searchParams.get('mediaId') ?? '0')
  const mediaType = searchParams.get('mediaType') ?? ''
  const comments = await prisma.comment.findMany({
    where: { mediaId, mediaType, approved: true },
    include: { author: { select: { name: true, image: true } } },
    orderBy: { createdAt: 'desc' },
  })
  return NextResponse.json(comments)
}

export async function POST(req: NextRequest) {
```

```tsx
'use client'
import { useState, useEffect, useCallback } from 'react'
import { Search } from 'lucide-react'
import MediaCard from '@/components/media/MediaCard'

interface SearchResult {
  id: number
  title?: string
  name?: string
  poster_path: string | null
  vote_average: number
  release_date?: string
  first_air_date?: string
  original_language: string
  media_type: 'movie' | 'tv'
}

export default function RecherchePage() {
  const [query, setQuery] = useState('')
  const [results, setResults] = useState<SearchResult[]>([])
  const [loading, setLoading] = useState(false)

  const search = useCallback(async (q: string) => {
    if (!q.trim()) { setResults([]); return }
    setLoading(true)
    try {
      const res = await fetch(`/api/search?q=${encodeURIComponent(q)}`)
      const data = await res.json()
      setResults(data.results?.filter((r: SearchResult) => r.media_type === 'movie' || r.media_type === 'tv') ?? [])
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => {
    const timer = setTimeout(() => search(query), 400)
    return () => clearTimeout(timer)
  }, [query, search])

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold">Recherche</h1>
      <div className="relative">
        <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-zinc-500" size={20} />
        <input
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Rechercher un film ou une série…"
          className="w-full bg-surface border border-zinc-700 rounded-xl pl-10 pr-4 py-3 text-white placeholder-zinc-500 focus:outline-none focus:border-zinc-500 transition"
          autoFocus
        />
      </div>
      {loading && <p className="text-zinc-500 text-sm">Recherche en cours…</p>}
      {results.length > 0 && (
        <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 gap-4">
          {results.map((r) => (
            <MediaCard
              key={`${r.media_type}-${r.id}`}
              item={{ ...r, genre_ids: [] }}
              type={r.media_type}
            />
          ))}

        <p className="text-zinc-500 text-center py-12">Aucun résultat pour &ldquo;{query}&rdquo;</p>
```

```tsx
interface Props {
  params: Promise<{ slug: string }>
  searchParams: Promise<{ page?: string }>
}

export async function generateMetadata({ params }: Props): Promise<Metadata> {
  const { slug } = await params
  const genreId = parseInt(slug.split('-')[0])
  const genre = TMDB_GENRES.find((g) => g.id === genreId)
  if (!genre) return {}
  return { title: `Films ${genre.name} en streaming` }
}

export default async function GenrePage({ params, searchParams }: Props) {
  const { slug } = await params
  const sp = await searchParams
  const genreId = parseInt(slug.split('-')[0])
  const genre = TMDB_GENRES.find((g) => g.id === genreId)
  if (!genre) notFound()

  const page = parseInt(sp.page ?? '1')
  const data = await discoverByGenre('movie', genreId, { page })
  const totalPages = Math.min(data.total_pages, 500)

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold">Films {genre.name}</h1>
      <MediaGrid items={data.results} type="movie" />
      <div className="flex items-center justify-center gap-2 py-4">
        {page > 1 && (
          <Link href={`/genre/${slug}?page=${page - 1}`} className="px-4 py-2 bg-surface rounded-lg hover:bg-zinc-700 transition text-sm">
            ← Précédent
          </Link>
        )}
        <span className="text-zinc-400 text-sm">Page {page} / {totalPages}</span>
        {page < totalPages && (
          <Link href={`/genre/${slug}?page=${page + 1}`} className="px-4 py-2 bg-surface rounded-lg hover:bg-zinc-700 transition text-sm">
            Suivant →
          </Link>
        )}
      </div>
    </div>
  )
}
```

```ts
// app/sitemap.ts
import { MetadataRoute } from 'next'
import { getPopular } from '@/lib/tmdb'
import { slugify } from '@/lib/utils'

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const base = process.env.NEXTAUTH_URL ?? 'https://streamvf.vercel.app'

  const [movies, shows] = await Promise.all([
    getPopular('movie', 1),
    getPopular('tv', 1),
  ])

  const movieUrls = movies.results.map((m) => ({
    url: `${base}/films/${m.id}/${slugify((m as any).title ?? '')}`,
    lastModified: new Date(),
    changeFrequency: 'monthly' as const,
    priority: 0.7,
  }))

  const tvUrls = shows.results.map((s) => ({
    url: `${base}/series/${s.id}/${slugify((s as any).name ?? '')}`,
    lastModified: new Date(),
    changeFrequency: 'weekly' as const,
    priority: 0.7,
  }))

  return [
    { url: base, lastModified: new Date(), changeFrequency: 'daily', priority: 1 },
    { url: `${base}/films`, lastModified: new Date(), changeFrequency: 'daily', priority: 0.9 },
    { url: `${base}/series`, lastModified: new Date(), changeFrequency: 'daily', priority: 0.9 },
    ...movieUrls,
    ...tvUrls,
  ]
}
```
